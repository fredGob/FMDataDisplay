# Colonnes de notes ajoutées aux tableaux du jeu (plugin FM27 Explorer).
# Fichier relu au lancement du jeu. Avec touches_dev=1 (fm27explorer_notes.txt) : F10 = relire ce fichier et les notes ;
# F11 = ajouter toutes les colonnes au tableau affiché ; Maj+F11 / Ctrl+F11 = seuil du filtre +5 / -5.
actif=1
# proprietes=1 (depuis 0.12.0) : chaque colonne a sa propre propriété (fm27note, fm27notegk…) ; la colonne modèle
# ne sert qu'à copier l'apparence (cellule, largeur) et la vraie colonne du jeu garde ses vraies valeurs.
# proprietes=0 : ancien mode (les valeurs de la colonne modèle sont remplacées partout, fiches comprises).
proprietes=1
# Catégorie du menu « insérer une colonne »
categorie=Notes FM27
# colonne=identifiant|titre|champ du fichier fm27_notes.tsv|colonne modèle du jeu|infobulle
# Le modèle est une colonne du jeu de type nombre entier, triée par l'interface. L'identifiant ne doit pas changer :
# il est enregistré dans les vues du jeu (« Notes FM27 », « CalculAuto »).
colonne=fm27-note|Note|note|international-youth-goals|Note FM27 : meilleure note parmi les profils du joueur (0-100)
colonne=fm27-note-gk|Note GB|GK|international-youth-caps|Note FM27 au profil gardien de but (0-100)
colonne=fm27-note-cb|Note DC|CB|loan-goals|Note FM27 au profil défenseur central (0-100)
colonne=fm27-note-fb|Note LAT|FB|loan-assists|Note FM27 au profil latéral (0-100)
colonne=fm27-note-dm|Note MDC|DM|loan-apps|Note FM27 au profil milieu défensif (0-100)
colonne=fm27-note-cm|Note MC|CM|international-goals-season|Note FM27 au profil milieu central (0-100)
colonne=fm27-note-am|Note MOC|AM|international-caps-season|Note FM27 au profil milieu offensif (0-100)
colonne=fm27-note-wg|Note AIL|WG|stats-career-league-goals|Note FM27 au profil ailier (0-100)
colonne=fm27-note-st|Note BT|ST|stats-career-league-appearances|Note FM27 au profil buteur (0-100)
